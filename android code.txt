// ==========================================================
// LOGIN USER
// ==========================================================

function loginUser(
  userId,
  password
) {

  const spreadsheet =
    SpreadsheetApp.openById(
      SPREADSHEET_ID
    );

  const sheet =
    spreadsheet.getSheetByName(
      USERS_SHEET_NAME
    );


  if (!sheet) {

    return jsonResponse({
      success: false,
      message:
        "Users sheet not found"
    });

  }


  const data =
    sheet.getDataRange().getValues();


  for (
    let i = 1;
    i < data.length;
    i++
  ) {

    const row =
      data[i];

    const sheetUserId =
      String(row[0]).trim();

    const sheetPassword =
      String(row[3]).trim();

    const sheetRole =
      String(row[4]).trim();

    const sheetStatus =
      String(row[5]).trim();


    if (
      sheetUserId === userId &&
      sheetPassword === password
    ) {

      if (
        sheetStatus.toLowerCase() !==
        "active"
      ) {

        return jsonResponse({
          success: false,
          message:
            "Account is not active"
        });

      }


      return jsonResponse({

        success: true,

        message:
          "Login successful",

        userId:
          sheetUserId,

        role:
          sheetRole

      });

    }

  }


  return jsonResponse({

    success: false,

    message:
      "Invalid User ID or password"

  });

}

// REGISTER USER
// ==========================================================

function registerUser(
  userId,
  name,
  email,
  mobile,
  password,
  role,
  github,
  linkedin
) {

  const spreadsheet =
    SpreadsheetApp.openById(
      SPREADSHEET_ID
    );

  const sheet =
    spreadsheet.getSheetByName(
      USERS_SHEET_NAME
    );


  if (!sheet) {

    return jsonResponse({
      success: false,
      message:
        "Users sheet not found"
    });

  }


  const data =
    sheet.getDataRange().getValues();


  for (
    let i = 1;
    i < data.length;
    i++
  ) {

    if (
      String(data[i][0])
        .trim()
        .toLowerCase() ===
      userId.toLowerCase()
    ) {

      return jsonResponse({
        success: false,
        message:
          "User ID already exists"
      });

    }

  }


  for (
    let i = 1;
    i < data.length;
    i++
  ) {

    if (
      String(data[i][2])
        .trim()
        .toLowerCase() ===
      email.toLowerCase()
    ) {

      return jsonResponse({
        success: false,
        message:
          "Email already exists"
      });

    }

  }


  sheet.appendRow([

    userId,
    name,
    email,
    password,
    role,
    "Active"

  ]);


  return jsonResponse({

    success: true,
    message:
      "Registration successful",
    userId:
      userId,
    role:
      role.toLowerCase()

  });

}

function receiveResumeChunk(e) {

    const uploadId =

        e.parameter.uploadId || ""
      ).trim();

    const userId =
      String(
        e.parameter.userId || ""
      ).trim();

    const chunkIndex =
      Number(
        e.parameter.chunkIndex || 0
      );

    const totalChunks =
      Number(
        e.parameter.totalChunks || 0
      );

    const fileName =
      String(
        e.parameter.fileName || ""
      ).trim();

    const chunk =
      String(
        e.parameter.chunk || ""
      );


    if (

    }


    const spreadsheet =
      SpreadsheetApp.openById(

      spreadsheet.getSheetByName(
        "_ResumeChunks"
      );


    if (!sheet) {

      sheet =
        spreadsheet.insertSheet(
          "_ResumeChunks"
        );

      sheet.appendRow([

        "Upload ID",
        "User ID",
        "File Name",
        "Chunk Index",
        "Total Chunks",
        "Chunk Data",
        "Created"

      ]);

    }


    sheet.appendRow([

      uploadId,
      userId,
      fileName,
      chunkIndex,

      new Date()

      success: true,

      success: false,
      message:
        "Chunk upload failed",
      details:
        error.toString()

    });

  }

// ==========================================================
// FINISH RESUME UPLOAD
// ==========================================================

function finishResumeUpload(e) {

  try {

    const uploadId =
      String(
        e.parameter.uploadId || ""
      ).trim();

    const userId =
      String(
        e.parameter.userId || ""
      ).trim();

    const fileName =
      String(
        e.parameter.fileName || ""
      ).trim();

    const mimeType =
      String(
        e.parameter.mimeType ||
        "application/pdf"
      ).trim();

    const totalChunks =
      Number(
        e.parameter.totalChunks || 0
      );


    if (
      uploadId === "" ||
      userId === "" ||
      fileName === "" ||
      totalChunks <= 0
    ) {

      return jsonResponse({

        success: false,
        message:
          "Invalid finish upload data"

      });

    }


    const spreadsheet =
      SpreadsheetApp.openById(
        SPREADSHEET_ID
      );


    const chunkSheet =
      spreadsheet.getSheetByName(
        "_ResumeChunks"
      );


    if (!chunkSheet) {

      return jsonResponse({

        success: false,
        message:
          "Resume chunk sheet not found"

      });

    }


    const data =
      chunkSheet
        .getDataRange()
        .getValues();


    const chunks = [];


    for (
      let i = 1;
      i < data.length;
      i++
    ) {

      const row =
        data[i];

      const rowUploadId =
        String(row[0]).trim();


      if (
        rowUploadId === uploadId
      ) {

        chunks.push({

          index:
            Number(row[3]),

          data:
            String(row[5]).trim()

        });

      }

    }


    if (
      chunks.length !== totalChunks
    ) {

      return jsonResponse({

        success: false,
        message:
          "Not all resume chunks were received",
        received:
          chunks.length,
        expected:
          totalChunks

      });

    }


    chunks.sort(
      function(a, b) {

        return a.index - b.index;

      }
    );


    for (
      let i = 0;
      i < chunks.length;
      i++
    ) {

      if (
        chunks[i].index !== i + 1
      ) {

        return jsonResponse({

          success: false,
          message:
            "Resume chunks are missing or out of order",
          expectedChunk:
            i + 1,
          receivedChunk:
            chunks[i].index

        });

      }

    }


    const decodedChunks = [];

    let totalBytes = 0;


    for (
      let i = 0;
      i < chunks.length;
      i++
    ) {

      let chunkBase64 =
        String(
          chunks[i].data || ""
        ).trim();


      chunkBase64 =
        chunkBase64.replace(
          /\s/g,
          ""
        );


      if (
        chunkBase64.indexOf(
          "base64,"
        ) !== -1
      ) {

        chunkBase64 =
          chunkBase64.substring(
            chunkBase64.indexOf(
              "base64,"
            ) + 7
          );

      }


      if (
        chunkBase64 === ""
      ) {

        return jsonResponse({

          success: false,
          message:
            "Empty Base64 data in chunk " +
            (i + 1)

        });

      }


      const remainder =
        chunkBase64.length % 4;


      if (
        remainder === 2
      ) {

        chunkBase64 += "==";

      } else if (
        remainder === 3
      ) {

        chunkBase64 += "=";

      } else if (
        remainder === 1
      ) {

        return jsonResponse({

          success: false,
          message:
            "Invalid Base64 length in chunk " +
            (i + 1)

        });

      }


      let decodedChunk;


      try {

        decodedChunk =
          Utilities.base64DecodeWebSafe(
            chunkBase64
          );

      } catch (decodeError) {

        return jsonResponse({

          success: false,
          message:
            "Base64 decoding failed for chunk " +
            (i + 1),
          details:
            decodeError.toString()

        });

      }


      decodedChunks.push(
        decodedChunk
      );

      totalBytes +=
        decodedChunk.length;

    }


    const fileBytes =
      new Array(
        totalBytes
      );

    let bytePosition = 0;


    for (
      let i = 0;
      i < decodedChunks.length;
      i++
    ) {

      const currentChunk =
        decodedChunks[i];


      for (
        let j = 0;
        j < currentChunk.length;
        j++
      ) {

        fileBytes[
          bytePosition
        ] =
          currentChunk[j];

        bytePosition++;

      }

    }


    if (
      fileBytes.length === 0
    ) {

      return jsonResponse({

        success: false,
        message:
          "Decoded resume file is empty"

      });

    }


    const blob =
      Utilities.newBlob(
        fileBytes,
        mimeType,
        fileName
      );


    const folder =
      DriveApp.getFolderById(
        RESUME_FOLDER_ID
      );


    const file =
      folder.createFile(
        blob
      );


    const driveUrl =
      file.getUrl();


    let resumeSheet =
      spreadsheet.getSheetByName(
        RESUMES_SHEET_NAME
      );


    if (!resumeSheet) {

      resumeSheet =
        spreadsheet.insertSheet(
          RESUMES_SHEET_NAME
        );

      resumeSheet.appendRow([

        "User ID",
        "File Name",
        "Drive URL",
        "Upload Date",
        "Status"

      ]);

    }


    resumeSheet.appendRow([

      userId,
      fileName,
      driveUrl,
      new Date(),
      "Uploaded"

    ]);


    for (
      let i = data.length - 1;
      i >= 1;
      i--
    ) {

      if (
        String(data[i][0]).trim() ===
        uploadId
      ) {

        chunkSheet.deleteRow(
          i + 1
        );

      }

    }


    return jsonResponse({

      success: true,
      message:
        "Resume uploaded successfully",
      userId:
        userId,
      fileName:
        fileName,
      driveUrl:
        driveUrl,
      fileSize:
        fileBytes.length

    });


  } catch (error) {

    return jsonResponse({

      success: false,
      message:
        "Resume finalization failed",
      details:
        error.toString(),
      stack:
        error.stack || ""

    });

  }

}

function jsonResponse(data) {

  return ContentService

    .createTextOutput(
      JSON.stringify(data)

      ContentService.MimeType.JSON
    );

function handleGetJobs(ss, params) {

  try {

      ss.getSheetByName(

      );


    if (!sheet) {

      return jsonResponse({
        success: true,
        jobs: []
      });

      sheet.getDataRange()
        .getValues();


    if (
      data.length <= 1
    ) {

      return jsonResponse({
        success: true,
        jobs: []
      });

    }


    const headers =
      data[0].map(
        function(header) {

          return String(header)
            .trim()
            .toLowerCase();

        }
      );


    const jobs = [];


    for (
      let i = 1;
      i < data.length;
      i++
    ) {

      const row =
        data[i];


      const getValue =
        function(names) {

          for (
            let k = 0;
            k < names.length;
            k++
          ) {

            const index =
              headers.indexOf(
                names[k]
                  .toLowerCase()
              );


            if (
              index !== -1 &&
              row[index] !== undefined &&
              row[index] !== null
            ) {

              return String(
                row[index]
              ).trim();

            }

          }


          return "";

        };


      const jobId =
        getValue([
          "job id",
          "jobid",
          "id"
        ]);

      const title =
        getValue([
          "job title",
          "jobtitle",
          "title"
        ]);

      const company =
        getValue([
          "company name",
          "companyname",
          "company"
        ]);

      const location =
        getValue([
          "location"
        ]);

      const jobType =
        getValue([
          "job type",
          "jobtype",
          "type"
        ]);

      const skills =
        getValue([
          "required skills",
          "requiredskills",
          "skills",
          "skillsrequired"
        ]);

      const salary =
        getValue([
          "salary",
          "stipend"
        ]);

      const description =
        getValue([
          "description",
          "job description",
          "jobdescription"
        ]);

      const status =
        getValue([
          "status",
          "job status",
          "jobstatus"
        ]);

      const postedDate =
        getValue([
          "posted date",
          "posteddate",
          "date",
          "created at",
          "createdat"
        ]);


      const statusLower =
        status.toLowerCase();


      if (
        statusLower !== "" &&
        statusLower !== "active"
      ) {

        continue;

      }


      if (
        jobId !== "" ||
        title !== ""
      ) {

        jobs.push({

          jobId:
            jobId ||
            ("JOB" + i),

          title:
            title,

          company:
            company,

          location:
            location,

          jobType:
            jobType,

          skills:
            skills,

          salary:
            salary,

          description:
            description,

          status:
            status ||
            "Active",

          postedDate:
            postedDate

        });

      }

    }


    return jsonResponse({

      success: true,
      jobs:
        jobs

    });


  } catch (error) {

    return jsonResponse({

      success: false,
      message:
        "Error fetching jobs",
      details:
        error.toString()

    });

  }

function applyForJob(e) {

  try {

    const userId =
      String(
        e.parameter.userId || ""
      ).trim();

    const jobId =
      String(
        e.parameter.jobId || ""
      ).trim();


    if (
      userId === "" ||
      jobId === ""
    ) {

      return jsonResponse({
        success: false,
        message:
          "User ID and Job ID are required"
      });

    }


    const spreadsheet =
      SpreadsheetApp.openById(
        SPREADSHEET_ID
      );


    let sheet =
      spreadsheet.getSheetByName(
        APPLICATIONS_SHEET_NAME
      );


    if (!sheet) {

      sheet =
        spreadsheet.insertSheet(
          APPLICATIONS_SHEET_NAME
        );

      sheet.appendRow([
        "Application ID",
        "Job ID",
        "User ID",
        "Applied Date",
        "Status"
      ]);

    }


    const data =
      sheet.getDataRange()
        .getValues();


    for (
      let i = 1;
      i < data.length;
      i++
    ) {

      const existingJobId =
        String(
          data[i][1] || ""
        ).trim();

      const existingUserId =
        String(
          data[i][2] || ""
        ).trim();


      if (
        existingJobId.toLowerCase() ===
          jobId.toLowerCase() &&
        existingUserId.toLowerCase() ===
          userId.toLowerCase()
      ) {

        return jsonResponse({

          success: false,

          message:
            "You have already applied for this job"

        });

      }

    }


    const applicationId =
      "APP" +
      new Date().getTime();


    sheet.appendRow([

      applicationId,
      jobId,
      userId,
      new Date(),
      "Applied"

    ]);

    // ========================================================
    // PHASE 8 - CREATE APPLICATION NOTIFICATION
    // ========================================================

    addNotification(
      spreadsheet,
      userId,
      "Application Submitted",
      "You have successfully applied for a job.",
      "application"
    );


    return jsonResponse({

      success: true,

      message:
        "Application submitted successfully",

      applicationId:
        applicationId,

      status:
        "Applied"

    });


  } catch (error) {

    return jsonResponse({

      success: false,

      message:
        "Job application failed",

      details:
        error.toString()

    });

  }

}

function getNotifications(e) {

  try {

    const userId =
      String(
        e.parameter.userId || ""
      ).trim();


    if (userId === "") {

      return jsonResponse({

        success: false,
        message:
          "User ID is required"

      });

    }


    const spreadsheet =
      SpreadsheetApp.openById(
        SPREADSHEET_ID
      );


    let sheet =
      spreadsheet.getSheetByName(
        "Notifications"
      );


    // Create the sheet automatically
    // if it does not exist yet.
    if (!sheet) {

      sheet =
        spreadsheet.insertSheet(
          "Notifications"
        );

      sheet.appendRow([

        "Notification ID",
        "User ID",
        "Title",
        "Message",
        "Type",
        "Created Date",
        "Is Read"

      ]);


      return jsonResponse({

        success: true,
        notifications: [],
        unreadCount: 0

      });

    }


    const data =
      sheet
        .getDataRange()
        .getValues();


    const notifications = [];
    let unreadCount = 0;


    for (
      let i = 1;
      i < data.length;
      i++
    ) {

      const row =
        data[i];


      const rowUserId =
        String(
          row[1] || ""
        ).trim();


      if (
        rowUserId.toLowerCase() !==
        userId.toLowerCase()
      ) {

        continue;

      }


      const isRead =
        row[6] === true ||
        String(
          row[6] || ""
        ).trim().toLowerCase() === "true" ||
        String(
          row[6] || ""
        ).trim().toLowerCase() === "read";


      if (!isRead) {

        unreadCount++;

      }


      notifications.push({

        notificationId:
          String(
            row[0] || ""
          ),

        userId:
          rowUserId,

        title:
          String(
            row[2] || ""
          ),

        message:
          String(
            row[3] || ""
          ),

        type:
          String(
            row[4] || "general"
          ),

        createdAt:
          row[5] instanceof Date
            ? row[5].toISOString()
            : String(
                row[5] || ""
              ),

        isRead:
          isRead

      });

    }


    // Newest notifications first.
    notifications.reverse();


    return jsonResponse({

      success: true,

      notifications:
        notifications,

      unreadCount:
        unreadCount

    });


  } catch (error) {

    return jsonResponse({

      success: false,

      message:
        "Failed to fetch notifications",

      details:
        error.toString()

    });

  }

}

              salary:
                getJobVal(
                  jRow,
                  [
                    "salary",
                    "stipend"
                  ]
                )
            };

            if (
              !companyId ||
              companyId === "all" ||
              jComp
                .toLowerCase()
                .indexOf(companyId) !== -1 ||
              companyId
                .indexOf(
                  jComp.toLowerCase()
                ) !== -1
            ) {

              companyJobIds.push(
                jId.toLowerCase()
              );

            }

          }

        }

      }

    }

    var profilesMap = {};

    var profSheet =
      ss.getSheetByName(
        "StudentProfiles"
      );

    if (profSheet) {

      var profData =
        profSheet
          .getDataRange()
          .getValues();

      if (profData.length > 1) {

        var profHeaders =
          profData[0].map(
            function(h) {
              return h.toString()
                .trim()
                .toLowerCase();
            }
          );

        var getProfVal =
          function(row, names) {

            for (
              var k = 0;
              k < names.length;
              k++
            ) {

              var idx =
                profHeaders.indexOf(
                  names[k].toLowerCase()
                );

              if (
                idx !== -1 &&
                row[idx] !== undefined &&
                row[idx] !== null
              ) {

                return row[idx]
                  .toString()
                  .trim();

              }

            }

            return "";
          };

        for (
          var p = 1;
          p < profData.length;
          p++
        ) {

          var pRow =
            profData[p];

          var pUserId =
            getProfVal(
              pRow,
              [
                "userid",
                "user id",
                "studentid",
                "student id"
              ]
            );

          if (pUserId) {

            profilesMap[
              pUserId.toLowerCase()
            ] = {

              fullName:
                getProfVal(
                  pRow,
                  [
                    "fullname",
                    "full name",
                    "name"
                  ]
                ),

              cgpa:
                getProfVal(
                  pRow,
                  ["cgpa", "c.g.p.a"]
                )
            };

          }

        }

      }

    }

    var usersMap = {};

    var usersSheet =
      ss.getSheetByName("Users");

    if (usersSheet) {

      var usersData =
        usersSheet
          .getDataRange()
          .getValues();

      if (usersData.length > 1) {

        var userHeaders =
          usersData[0].map(
            function(h) {
              return h.toString()
                .trim()
                .toLowerCase();
            }
          );

        var getUserVal =
          function(row, names) {

            for (
              var k = 0;
              k < names.length;
              k++
            ) {

              var idx =
                userHeaders.indexOf(
                  names[k].toLowerCase()
                );

              if (
                idx !== -1 &&
                row[idx] !== undefined &&
                row[idx] !== null
              ) {

                return row[idx]
                  .toString()
                  .trim();

              }

            }

            return "";
          };

        for (
          var u = 1;
          u < usersData.length;
          u++
        ) {

          var uRow =
            usersData[u];

          var uId =
            getUserVal(
              uRow,
              [
                "userid",
                "user id"
              ]
            );

          if (uId) {

            usersMap[
              uId.toLowerCase()
            ] = {

              name:
                getUserVal(
                  uRow,
                  [
                    "name",
                    "fullname",
                    "full name"
                  ]
                ),

              email:
                getUserVal(
                  uRow,
                  ["email"]
                )
            };

          }

        }

      }

    }

    var applicationsSheet =
      ss.getSheetByName(
        "Applications"
      );

    var applications = [];

    if (
      !applicationsSheet
    ) {

      return jsonResponse({
        success: true,
        applications: []
      });

    }

    var appData =
      applicationsSheet
        .getDataRange()
        .getValues();

    if (appData.length <= 1) {

      return jsonResponse({
        success: true,
        applications: []
      });

    }

    var appHeaders =
      appData[0].map(
        function(h) {
          return h.toString()
            .trim()
            .toLowerCase();
        }
      );

    var getAppVal =
      function(row, names) {

        for (
          var k = 0;
          k < names.length;
          k++
        ) {

          var idx =
            appHeaders.indexOf(
              names[k].toLowerCase()
            );

          if (
            idx !== -1 &&
            row[idx] !== undefined &&
            row[idx] !== null
          ) {

            return row[idx]
              .toString()
              .trim();

          }

        }

        return "";
      };

    for (
      var a = 1;
      a < appData.length;
      a++
    ) {

      var row =
        appData[a];

      var jobId =
        getAppVal(
          row,
          [
            "jobid",
            "job id"
          ]
        );

      if (
        !jobId ||
        companyJobIds.indexOf(
          jobId.toLowerCase()
        ) === -1
      ) {

        continue;

      }

      var userId =
        getAppVal(
          row,
          [
            "userid",
            "user id"
          ]
        );

      var profile =
        profilesMap[
          userId.toLowerCase()
        ] || {};

      var user =
        usersMap[
          userId.toLowerCase()
        ] || {};

      var job =
        jobsMap[
          jobId.toLowerCase()
        ] || {};

      applications.push({

        applicationId:
          getAppVal(
            row,
            [
              "applicationid",
              "application id",
              "appid"
            ]
          ),

        jobId: jobId,

        userId: userId,

        studentName:
          profile.fullName ||
          user.name ||
          userId,

        email:
          user.email || "",

        cgpa:
          profile.cgpa || "",

        title:
          job.title || "",

        company:
          job.company || "",

        location:
          job.location || "",

        salary:
          job.salary || "",

        appliedDate:
          getAppVal(
            row,
            [
              "applieddate",
              "applied date",
              "date"
            ]
          ),

        status:
          getAppVal(
            row,
            [
              "status",
              "applicationstatus"
            ]
          ) || "Applied"

      });

    }

    return jsonResponse({
      success: true,
      applications: applications
    });

  } catch (error) {

    return jsonResponse({

      success: false,

      message:
        "Failed to fetch company applications",

      details:
        error.toString()

    });

  }

}

function handleUpdateApplicationStatus(ss, params) {

  try {

    var applicationId =
      String(
        params.applicationId ||
        params.applicationid ||
        ""
      ).trim();

    var status =
      String(
        params.status ||
        ""
      ).trim();

    var userId =
      String(
        params.userId ||
        params.userid ||
        ""
      ).trim();

    if (!applicationId || !status) {

      return jsonResponse({
        success: false,
        message: "Application ID and status are required"
      });

    }

    var sheet =
      ss.getSheetByName("Applications");

    if (!sheet) {

      return jsonResponse({
        success: false,
        message: "Applications sheet not found"
      });

    }

    var data =
      sheet.getDataRange().getValues();

    if (data.length <= 1) {

      return jsonResponse({
        success: false,
        message: "Application not found"
      });

    }

    var headers =
      data[0].map(function(h) {
        return String(h)
          .trim()
          .toLowerCase();
      });

    var applicationIdCol =
      headers.indexOf("application id");

    if (applicationIdCol === -1) {
      applicationIdCol =
        headers.indexOf("applicationid");
    }

    if (applicationIdCol === -1) {
      applicationIdCol = 0;
    }

    var statusCol =
      headers.indexOf("status");

    if (statusCol === -1) {
      statusCol = 4;
    }

    var userIdCol =
      headers.indexOf("user id");

    if (userIdCol === -1) {
      userIdCol =
        headers.indexOf("userid");
    }

    if (userIdCol === -1) {
      userIdCol = 2;
    }

    for (
      var i = 1;
      i < data.length;
      i++
    ) {

      var row =
        data[i];

      var rowApplicationId =
        String(
          row[applicationIdCol] || ""
        ).trim();

      if (
        rowApplicationId.toLowerCase() ===
        applicationId.toLowerCase()
      ) {

        row[statusCol] =
          status;

        sheet
          .getRange(
            i + 1,
            statusCol + 1
          )
          .setValue(status);

        var studentUserId =
          String(
            row[userIdCol] || userId || ""
          ).trim();

        // Create notification for student
        if (studentUserId) {

          addNotification(
            ss,
            studentUserId,
            "Application Status Updated",
            "Your application status has been updated to " + status + ".",
            "application"
          );

        }

        return jsonResponse({

          success: true,

          message:
            "Application status updated successfully",

          applicationId:
            applicationId,

          status:
            status

        });

      }

    }

    return jsonResponse({

      success: false,

      message:
        "Application not found"

    });

  } catch (error) {

    return jsonResponse({

      success: false,

      message:
        "Failed to update application status",

      details:
        error.toString()

    });

  }

}
