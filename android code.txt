// ==========================================================
// LOGIN USER
// ==========================================================

function loginUser(
  userId,
  password
) {

  const spreadsheet =
    SpreadsheetApp.openById(
      SPREADSHEET_ID
    );

  const sheet =
    spreadsheet.getSheetByName(
      USERS_SHEET_NAME
    );


  if (!sheet) {

    return jsonResponse({
      success: false,
      message:
        "Users sheet not found"
    });

  }


  const data =
    sheet.getDataRange().getValues();


  for (
    let i = 1;
    i < data.length;
    i++
  ) {

    const row =
      data[i];

    const sheetUserId =
      String(row[0]).trim();

    const sheetPassword =
      String(row[3]).trim();

    const sheetRole =
      String(row[4]).trim();

    const sheetStatus =
      String(row[5]).trim();


    if (
      sheetUserId === userId &&
      sheetPassword === password
    ) {

      if (
        sheetStatus.toLowerCase() !==
        "active"
      ) {

        return jsonResponse({
          success: false,
          message:
            "Account is not active"
        });

      }


      return jsonResponse({

        success: true,

        message:
          "Login successful",

        userId:
          sheetUserId,

        role:
          sheetRole

      });

    }

  }


  return jsonResponse({

    success: false,

    message:
      "Invalid User ID or password"

  });

}

// ==========================================================

function registerUser(
  userId,
  name,
  email,
  mobile,
  password,
  role,
  github,
  linkedin
) {

  const spreadsheet =
    SpreadsheetApp.openById(
      SPREADSHEET_ID
    );

  const sheet =
    spreadsheet.getSheetByName(
      USERS_SHEET_NAME
    );


  if (!sheet) {

    return jsonResponse({
      success: false,
      message:
        "Users sheet not found"
    });

  }


  const data =
    sheet.getDataRange().getValues();


  for (
    let i = 1;
    i < data.length;
    i++
  ) {

    if (
      String(data[i][0])
        .trim()
        .toLowerCase() ===
      userId.toLowerCase()
    ) {

      return jsonResponse({
        success: false,
        message:
          "User ID already exists"
      });

    }

  }


  for (
    let i = 1;
    i < data.length;
    i++
  ) {

    if (
      String(data[i][2])
        .trim()
        .toLowerCase() ===
      email.toLowerCase()
    ) {

      return jsonResponse({
        success: false,
        message:
          "Email already exists"
      });

    }

  }


  sheet.appendRow([

    userId,
    name,
    email,
    password,
    role,
    "Active"

  ]);


  return jsonResponse({

    success: true,
    message:
      "Registration successful",
    userId:
      userId,
    role:
      role.toLowerCase()

  });

}

// ==========================================================
// FINISH RESUME UPLOAD
// ==========================================================

function finishResumeUpload(e) {

  try {

    const uploadId =
      String(
        e.parameter.uploadId || ""
      ).trim();

    const userId =
      String(
        e.parameter.userId || ""
      ).trim();

    const fileName =
      String(
        e.parameter.fileName || ""
      ).trim();

    const mimeType =
      String(
        e.parameter.mimeType ||
        "application/pdf"
      ).trim();

    const totalChunks =
      Number(
        e.parameter.totalChunks || 0
      );


    if (
      uploadId === "" ||
      userId === "" ||
      fileName === "" ||
      totalChunks <= 0
    ) {

      return jsonResponse({

        success: false,
        message:
          "Invalid finish upload data"

      });

    }


    const spreadsheet =
      SpreadsheetApp.openById(
        SPREADSHEET_ID
      );


    const chunkSheet =
      spreadsheet.getSheetByName(
        "_ResumeChunks"
      );


    if (!chunkSheet) {

      return jsonResponse({

        success: false,
        message:
          "Resume chunk sheet not found"

      });

    }


    const data =
      chunkSheet
        .getDataRange()
        .getValues();


    const chunks = [];


    for (
      let i = 1;
      i < data.length;
      i++
    ) {

      const row =
        data[i];

      const rowUploadId =
        String(row[0]).trim();


      if (
        rowUploadId === uploadId
      ) {

        chunks.push({

          index:
            Number(row[3]),

          data:
            String(row[5]).trim()

        });

      }

    }


    if (
      chunks.length !== totalChunks
    ) {

      return jsonResponse({

        success: false,
        message:
          "Not all resume chunks were received",
        received:
          chunks.length,
        expected:
          totalChunks

      });

    }


    chunks.sort(
      function(a, b) {

        return a.index - b.index;

      }
    );


    for (
      let i = 0;
      i < chunks.length;
      i++
    ) {

      if (
        chunks[i].index !== i + 1
      ) {

        return jsonResponse({

          success: false,
          message:
            "Resume chunks are missing or out of order",
          expectedChunk:
            i + 1,
          receivedChunk:
            chunks[i].index

        });

      }

    }


    const decodedChunks = [];

    let totalBytes = 0;


    for (
      let i = 0;
      i < chunks.length;
      i++
    ) {

      let chunkBase64 =
        String(
          chunks[i].data || ""
        ).trim();


      chunkBase64 =
        chunkBase64.replace(
          /\s/g,
          ""
        );


      if (
        chunkBase64.indexOf(
          "base64,"
        ) !== -1
      ) {

        chunkBase64 =
          chunkBase64.substring(
            chunkBase64.indexOf(
              "base64,"
            ) + 7
          );

      }


      if (
        chunkBase64 === ""
      ) {

        return jsonResponse({

          success: false,
          message:
            "Empty Base64 data in chunk " +
            (i + 1)

        });

      }


      const remainder =
        chunkBase64.length % 4;


      if (
        remainder === 2
      ) {

        chunkBase64 += "==";

      } else if (
        remainder === 3
      ) {

        chunkBase64 += "=";

      } else if (
        remainder === 1
      ) {

        return jsonResponse({

          success: false,
          message:
            "Invalid Base64 length in chunk " +
            (i + 1)

        });

      }


      let decodedChunk;


      try {

        decodedChunk =
          Utilities.base64DecodeWebSafe(
            chunkBase64
          );

      } catch (decodeError) {

        return jsonResponse({

          success: false,
          message:
            "Base64 decoding failed for chunk " +
            (i + 1),
          details:
            decodeError.toString()

        });

      }


      decodedChunks.push(
        decodedChunk
      );

      totalBytes +=
        decodedChunk.length;

    }


    const fileBytes =
      new Array(
        totalBytes
      );

    let bytePosition = 0;


    for (
      let i = 0;
      i < decodedChunks.length;
      i++
    ) {

      const currentChunk =
        decodedChunks[i];


      for (
        let j = 0;
        j < currentChunk.length;
        j++
      ) {

        fileBytes[
          bytePosition
        ] =
          currentChunk[j];

        bytePosition++;

      }

    }


    if (
      fileBytes.length === 0
    ) {

      return jsonResponse({

        success: false,
        message:
          "Decoded resume file is empty"

      });

    }


    const blob =
      Utilities.newBlob(
        fileBytes,
        mimeType,
        fileName
      );


    const folder =
      DriveApp.getFolderById(
        RESUME_FOLDER_ID
      );


    const file =
      folder.createFile(
        blob
      );


    const driveUrl =
      file.getUrl();


    let resumeSheet =
      spreadsheet.getSheetByName(
        RESUMES_SHEET_NAME
      );


    if (!resumeSheet) {

      resumeSheet =
        spreadsheet.insertSheet(
          RESUMES_SHEET_NAME
        );

      resumeSheet.appendRow([

        "User ID",
        "File Name",
        "Drive URL",
        "Upload Date",
        "Status"

      ]);

    }


    resumeSheet.appendRow([

      userId,
      fileName,
      driveUrl,
      new Date(),
      "Uploaded"

    ]);


    for (
      let i = data.length - 1;
      i >= 1;
      i--
    ) {

      if (
        String(data[i][0]).trim() ===
        uploadId
      ) {

        chunkSheet.deleteRow(
          i + 1
        );

      }

    }


    return jsonResponse({

      success: true,
      message:
        "Resume uploaded successfully",
      userId:
        userId,
      fileName:
        fileName,
      driveUrl:
        driveUrl,
      fileSize:
        fileBytes.length

    });


  } catch (error) {

    return jsonResponse({

      success: false,
      message:
        "Resume finalization failed",
      details:
        error.toString(),
      stack:
        error.stack || ""

    });

  }

}

function postJob(e) {

    const companyName =
      String(
        e.parameter.companyName || ""
      ).trim();

    const jobTitle =
      String(
        e.parameter.jobTitle || ""
      ).trim();

    const location =
      String(
        e.parameter.location || ""
      ).trim();

    const skills =
      String(
        e.parameter.skills || ""
      ).trim();

    const salary =
      String(
        e.parameter.salary || ""
      ).trim();

    const description =
      String(
        e.parameter.description || ""
      ).trim();


    if (
      companyName === "" ||
      jobTitle === "" ||
      location === "" ||
      skills === "" ||
      description === ""
    ) {

      return jsonResponse({
        success: false,
        message:
          "Required job fields are missing"
      });

    }


    const spreadsheet =
      SpreadsheetApp.openById(
        SPREADSHEET_ID
      );


    let sheet =
      spreadsheet.getSheetByName(
        JOBS_SHEET_NAME
      );


    if (!sheet) {

      sheet =
        spreadsheet.insertSheet(
          JOBS_SHEET_NAME
        );

      sheet.appendRow([
        "Job ID",
        "Company Name",
        "Job Title",
        "Location",
        "Required Skills",
        "Salary",
        "Description",
        "Posted Date",
        "Status"
      ]);

    }


    const jobId =
      "JOB" +
      new Date().getTime();


    const maxHiringRaw =
      String(
        e.parameter.maxHiring ||
        e.parameter.max_hiring ||
        e.parameter.maxNoOfHiring ||
        ""
      ).trim();

    const maxHiring =
      parseInt(maxHiringRaw, 10);

    const validMaxHiring =
      (!isNaN(maxHiring) && maxHiring > 0) ? maxHiring : "";

    const data =
      sheet.getDataRange().getValues();

    const headers =
      data.length > 0
        ? data[0].map(function(h) { return String(h).trim().toLowerCase(); })
        : [];

    let maxCol = headers.indexOf("max hiring");
    if (maxCol === -1) maxCol = headers.indexOf("maxhiring");
    if (maxCol === -1) maxCol = headers.indexOf("max_hiring");
    if (maxCol === -1) {
      sheet.getRange(1, headers.length + 1).setValue("Max Hiring");
      headers.push("max hiring");
      maxCol = headers.length - 1;
    }

    let appCol = headers.indexOf("application count");
    if (appCol === -1) appCol = headers.indexOf("applicationcount");
    if (appCol === -1) appCol = headers.indexOf("application_count");
    if (appCol === -1) {
      sheet.getRange(1, headers.length + 1).setValue("Application Count");
      headers.push("application count");
      appCol = headers.length - 1;
    }

    const newRow = new Array(headers.length);
    for (let c = 0; c < headers.length; c++) {
      newRow[c] = "";
    }

    const setColVal = function(names, val) {
      for (let k = 0; k < names.length; k++) {
        const idx = headers.indexOf(names[k].toLowerCase());
        if (idx !== -1) {
          newRow[idx] = val;
          return;
        }
      }
    };

    setColVal(["job id", "jobid", "id"], jobId);
    setColVal(["company name", "company", "companyname"], companyName);
    setColVal(["job title", "title", "jobtitle"], jobTitle);
    setColVal(["location"], location);
    setColVal(["required skills", "skills", "requiredskills"], skills);
    setColVal(["salary", "stipend"], salary);
    setColVal(["description"], description);
    setColVal(["posted date", "date"], new Date());
    setColVal(["status", "job status"], "Active");
    setColVal(["max hiring", "maxhiring", "max_hiring", "max no of hiring"], validMaxHiring);
    setColVal(["application count", "applicationcount", "application_count"], 0);

    sheet.appendRow(newRow);


    return jsonResponse({

      success: true,
      message:
        "Job posted successfully",
      jobId:
        jobId

    });


  } catch (error) {

    return jsonResponse({

      success: false,
      message:
        "Job posting failed",
      details:
        error.toString()

    });

  }

}

// PHASE 7 - APPLY FOR JOB
// ==========================================================

function applyForJob(e) {
  const lock = LockService.getScriptLock();
  let lockAcquired = false;

  try {
    lockAcquired = lock.tryLock(20000);
    if (!lockAcquired) {
      return jsonResponse({
        success: false,
        message: "Server is currently busy. Please try submitting your application again."
      });
    }

    const userId =
      String(
        e.parameter.userId || ""
      ).trim();

    const jobId =
      String(
        e.parameter.jobId || ""
      ).trim();

    if (
      userId === "" ||
      jobId === ""
    ) {
      return jsonResponse({
        success: false,
        message:
          "User ID and Job ID are required"
      });
    }

    const spreadsheet =
      SpreadsheetApp.openById(
        SPREADSHEET_ID
      );

    let sheet =
      spreadsheet.getSheetByName(
        APPLICATIONS_SHEET_NAME
      );

    if (!sheet) {
      sheet =
        spreadsheet.insertSheet(
          APPLICATIONS_SHEET_NAME
        );

      sheet.appendRow([
        "Application ID",
        "Job ID",
        "User ID",
        "Applied Date",
        "Status"
      ]);
    }

    const data =
      sheet.getDataRange()
        .getValues();

    for (
      let i = 1;
      i < data.length;
      i++
    ) {
      const existingJobId =
        String(
          data[i][1] || ""
        ).trim();

      const existingUserId =
        String(
          data[i][2] || ""
        ).trim();

      if (
        existingJobId.toLowerCase() ===
          jobId.toLowerCase() &&
        existingUserId.toLowerCase() ===
          userId.toLowerCase()
      ) {
        return jsonResponse({
          success: false,
          message:
            "You have already applied for this job"
        });
      }
    }

    // Check Jobs sheet for existence, status, and max hiring limit
    const jobsSheet =
      spreadsheet.getSheetByName(
        JOBS_SHEET_NAME
      );

    if (!jobsSheet) {
      return jsonResponse({
        success: false,
        message: "Job not found"
      });
    }

    const jData =
      jobsSheet.getDataRange().getValues();

    if (jData.length <= 1) {
      return jsonResponse({
        success: false,
        message: "Job not found"
      });
    }

    const jHeaders =
      jData[0].map(function(h) { return String(h).trim().toLowerCase(); });

    const findCol = function(names, fallback) {
      for (let k = 0; k < names.length; k++) {
        const idx = jHeaders.indexOf(names[k].toLowerCase());
        if (idx !== -1) return idx;
      }
      return fallback;
    };

    const jIdCol = findCol(["job id", "jobid", "id"], 0);
    const jStatusCol = findCol(["status", "job status", "jobstatus"], 8);

    let jMaxCol = findCol(["max hiring", "maxhiring", "max_hiring", "max no of hiring"], -1);
    if (jMaxCol === -1) {
      jobsSheet.getRange(1, jHeaders.length + 1).setValue("Max Hiring");
      jHeaders.push("max hiring");
      jMaxCol = jHeaders.length - 1;
    }

    let jCountCol = findCol(["application count", "applicationcount", "application_count", "applications"], -1);
    if (jCountCol === -1) {
      jobsSheet.getRange(1, jHeaders.length + 1).setValue("Application Count");
      jHeaders.push("application count");
      jCountCol = jHeaders.length - 1;
    }

    let targetJobRowIndex = -1;
    let targetJobRow = null;

    for (let j = 1; j < jData.length; j++) {
      const rowId = jData[j][jIdCol] ? String(jData[j][jIdCol]).trim().toLowerCase() : "";
      if (rowId === jobId.toLowerCase()) {
        targetJobRowIndex = j + 1;
        targetJobRow = jData[j];
        break;
      }
    }

    if (targetJobRowIndex === -1 || !targetJobRow) {
      return jsonResponse({
        success: false,
        message: "Job not found"
      });
    }

    const currentStatus = targetJobRow[jStatusCol] ? String(targetJobRow[jStatusCol]).trim() : "Active";
    const currentStatusLower = currentStatus.toLowerCase();

    const maxHiringRaw = (jMaxCol < targetJobRow.length && targetJobRow[jMaxCol] !== undefined && targetJobRow[jMaxCol] !== null)
      ? String(targetJobRow[jMaxCol]).trim()
      : "";
    const maxHiring = parseInt(maxHiringRaw, 10);
    const hasMaxHiring = !isNaN(maxHiring) && maxHiring > 0;

    const currentCountRaw = (jCountCol < targetJobRow.length && targetJobRow[jCountCol] !== undefined && targetJobRow[jCountCol] !== null)
      ? String(targetJobRow[jCountCol]).trim()
      : "0";
    let currentAppCount = parseInt(currentCountRaw, 10);
    if (isNaN(currentAppCount) || currentAppCount < 0) {
      currentAppCount = 0;
    }

    if (currentStatusLower !== "active") {
      if (currentStatusLower === "closed") {
        return jsonResponse({
          success: false,
          message: "Maximum number of hiring for this job has been reached. Applications are closed."
        });
      }
      return jsonResponse({
        success: false,
        message: "This job is currently inactive and not accepting applications"
      });
    }

    if (hasMaxHiring && currentAppCount >= maxHiring) {
      jobsSheet.getRange(targetJobRowIndex, jStatusCol + 1).setValue("Closed");
      return jsonResponse({
        success: false,
        message: "Maximum number of hiring for this job has been reached. Applications are closed."
      });
    }

    const applicationId =
      "APP" +
      new Date().getTime();

    sheet.appendRow([
      applicationId,
      jobId,
      userId,
      new Date(),
      "Applied"
    ]);

    const newAppCount = currentAppCount + 1;
    jobsSheet.getRange(targetJobRowIndex, jCountCol + 1).setValue(newAppCount);

    if (hasMaxHiring && newAppCount >= maxHiring) {
      jobsSheet.getRange(targetJobRowIndex, jStatusCol + 1).setValue("Closed");
    }

    addNotification(
      spreadsheet,
      userId,
      "Application Submitted",
      "You have successfully applied for a job.",
      "application"
    );

    return jsonResponse({
      success: true,
      message:
        "Application submitted successfully",
      applicationId:
        applicationId,
      status:
        "Applied"
    });

  } catch (error) {
    return jsonResponse({
      success: false,
      message:
        "Job application failed",
      details:
        error.toString()
    });
  } finally {
    if (lockAcquired) {
      try {
        lock.releaseLock();
      } catch (e) {}
    }
  }
}

      });

    }


    const data =
      sheet

    for (

      i < data.length;

        ).trim();

      ) {

        continue;

      }


      const isRead =
        row[6] === true ||
        String(
          row[6] || ""
        ).trim().toLowerCase() === "true" ||
        String(
          row[6] || ""
        ).trim().toLowerCase() === "read";


      if (!isRead) {

        unreadCount++;

      }


      notifications.push({

        notificationId:
          String(
            row[0] || ""
          ),

        userId:
          rowUserId,

        title:
          String(
            row[2] || ""
          ),

        message:
          String(
            row[3] || ""
          ),

        type:
          String(
            row[4] || "general"
          ),

        createdAt:
          row[5] instanceof Date
            ? row[5].toISOString()
            : String(
                row[5] || ""
              ),

        isRead:
          isRead

      });

    }


    // Newest notifications first.
    notifications.reverse();


    return jsonResponse({

      success: true,

      notifications:
        notifications,

      unreadCount:
        unreadCount

    });


  } catch (error) {

      success: false,

      message:

        error.toString()

    });

function handleUpdateApplicationStatus(ss, params) {

  try {

    var applicationId =
      String(
        params.applicationId ||
        params.applicationid ||
        ""
      ).trim();

    var status =
      String(
        params.status ||
        ""
      ).trim();

    var userId =
      String(
        params.userId ||
        params.userid ||
        ""
      ).trim();

    if (!applicationId || !status) {

      return jsonResponse({
        success: false,
        message: "Application ID and status are required"
      });

    }

    var sheet =
      ss.getSheetByName("Applications");

    if (!sheet) {

      return jsonResponse({
        success: false,
        message: "Applications sheet not found"
      });

    }

    var data =
      sheet.getDataRange().getValues();

    if (data.length <= 1) {

      return jsonResponse({
        success: false,
        message: "Application not found"
      });

    }

    var headers =
      data[0].map(function(h) {
        return String(h)
          .trim()
          .toLowerCase();
      });

    var applicationIdCol =
      headers.indexOf("application id");

    if (applicationIdCol === -1) {
      applicationIdCol =
        headers.indexOf("applicationid");
    }

    if (applicationIdCol === -1) {
      applicationIdCol = 0;
    }

    var statusCol =
      headers.indexOf("status");

    if (statusCol === -1) {
      statusCol = 4;
    }

    var userIdCol =
      headers.indexOf("user id");

    if (userIdCol === -1) {
      userIdCol =
        headers.indexOf("userid");
    }

    if (userIdCol === -1) {
      userIdCol = 2;
    }

    for (
      var i = 1;
      i < data.length;
      i++
    ) {

      var row =
        data[i];

      var rowApplicationId =
        String(
          row[applicationIdCol] || ""
        ).trim();

      if (
        rowApplicationId.toLowerCase() ===
        applicationId.toLowerCase()
      ) {

        row[statusCol] =
          status;

        sheet
          .getRange(
            i + 1,
            statusCol + 1
          )
          .setValue(status);

        var studentUserId =
          String(
            row[userIdCol] || userId || ""
          ).trim();

        // Create notification for student
        if (studentUserId) {

          addNotification(
            ss,
            studentUserId,
            "Application Status Updated",
            "Your application status has been updated to " + status + ".",
            "application"
          );

        }

        return jsonResponse({

          success: true,

          message:
            "Application status updated successfully",

          applicationId:
            applicationId,

          status:
            status

        });

      }

    }

    return jsonResponse({

      success: false,

      message:
        "Application not found"

    });

  } catch (error) {

    return jsonResponse({

      success: false,

      message:
        "Failed to update application status",

      details:
        error.toString()

    });

  }

}

function handleGetAdminStatistics(ss, params) {

        if (roleCol === -1) roleCol = uHeaders.indexOf("role name");

        for (var i = 1; i < uData.length; i++) {
          var uRow = uData[i];
          if (!uRow[0] && !uRow[1]) continue; // Skip empty rows

          var roleVal = (roleCol !== -1 && uRow[roleCol] !== undefined && uRow[roleCol] !== null)
            ? String(uRow[roleCol]).trim().toLowerCase()
            : "";

          if (roleVal === "student") {
            totalStudents++;
          }
        }
      }
    }

    // 2. Jobs Sheet - Count Total Active Jobs
    var jobsSheet = ss.getSheetByName("Jobs");
    if (jobsSheet) {
      var jData = jobsSheet.getDataRange().getValues();
      if (jData.length > 1) {
        var jHeaders = jData[0].map(function (h) { return h.toString().trim().toLowerCase(); });
        var statusCol = jHeaders.indexOf("status");
        if (statusCol === -1) statusCol = jHeaders.indexOf("job status");
        if (statusCol === -1) statusCol = jHeaders.indexOf("jobstatus");

        for (var j = 1; j < jData.length; j++) {
          var jRow = jData[j];
          if (!jRow[0] && !jRow[1]) continue; // Skip empty rows

          var jStatus = (statusCol !== -1 && jRow[statusCol] !== undefined && jRow[statusCol] !== null)
            ? String(jRow[statusCol]).trim().toLowerCase()
            : "active";

          if (!jStatus || jStatus === "active" || jStatus === "open") {
            totalJobs++;
          }
        }
      }
    }

    // 3. Applications Sheet - Count Applications, Shortlisted & Selected
    var appsSheet = ss.getSheetByName("Applications");
    if (appsSheet) {
      var aData = appsSheet.getDataRange().getValues();
      if (aData.length > 1) {
        var aHeaders = aData[0].map(function (h) { return h.toString().trim().toLowerCase(); });
        var aStatusCol = aHeaders.indexOf("status");

        if (aStatusCol === -1) aStatusCol = aHeaders.indexOf("applicationstatus");
        if (aStatusCol === -1) aStatusCol = 4; // Fallback to column index 4

        for (var k = 1; k < aData.length; k++) {
          var aRow = aData[k];
          if (!aRow[0] && !aRow[1] && !aRow[2]) continue; // Skip empty rows

            : "";

            selected++;
          }
        }
      }
    }

      ? parseFloat(((selected / totalApplications) * 100).toFixed(1))
      : 0.0;

      success: true,
      totalStudents: totalStudents,
      totalJobs: totalJobs,
      totalApplications: totalApplications,
      shortlisted: shortlisted,

      message: "Error generating admin statistics: " + err.toString()

// Action: update_job_status
function handleUpdateJobStatus(ss, params) {
  try {
    var jobIdReq = (params.jobId || params.id || "").toString().trim();
    var statusReq = (params.status || "").toString().trim();

    if (!jobIdReq) {
      return jsonResponse({ success: false, message: "Job ID is required" });
    }
    if (!statusReq) {
      return jsonResponse({ success: false, message: "Status is required" });
    }

    var normStatus = statusReq.toLowerCase();
    if (normStatus !== "active" && normStatus !== "inactive" && normStatus !== "closed") {
      return jsonResponse({ success: false, message: "Status must be 'Active', 'Inactive', or 'Closed'" });
    }
    var finalStatus = (normStatus === "active") ? "Active" : ((normStatus === "closed") ? "Closed" : "Inactive");

    var sheet = ss.getSheetByName(JOBS_SHEET_NAME || "Jobs");
    if (!sheet) {
      return jsonResponse({ success: false, message: "Jobs sheet not found" });
    }

    var data = sheet.getDataRange().getValues();
    if (data.length <= 1) {
      return jsonResponse({ success: false, message: "Job not found" });
    }

    var headers = data[0].map(function (h) { return h.toString().trim().toLowerCase(); });

    var jobIdCol = headers.indexOf("job id");
    if (jobIdCol === -1) jobIdCol = headers.indexOf("jobid");
    if (jobIdCol === -1) jobIdCol = headers.indexOf("id");
    if (jobIdCol === -1) jobIdCol = 0;

    var statusCol = headers.indexOf("status");
    if (statusCol === -1) statusCol = headers.indexOf("job status");
    if (statusCol === -1) statusCol = headers.indexOf("jobstatus");
    if (statusCol === -1) statusCol = 8;

    var titleCol = headers.indexOf("title");
    if (titleCol === -1) titleCol = headers.indexOf("job title");
    if (titleCol === -1) titleCol = headers.indexOf("jobtitle");
    if (titleCol === -1) titleCol = 1;

    var companyCol = headers.indexOf("company");
    if (companyCol === -1) companyCol = headers.indexOf("company name");
    if (companyCol === -1) companyCol = headers.indexOf("companyname");
    if (companyCol === -1) companyCol = 2;

    var rowIndex = -1;
    var foundJobId = "";
    var jobTitle = "";
    var companyName = "";

    for (var i = 1; i < data.length; i++) {
      var rJobId = data[i][jobIdCol] ? String(data[i][jobIdCol]).trim() : "";
      if (rJobId.toLowerCase() === jobIdReq.toLowerCase()) {
        rowIndex = i + 1;
        foundJobId = rJobId;
        jobTitle = data[i][titleCol] ? String(data[i][titleCol]).trim() : "Job";
        companyName = data[i][companyCol] ? String(data[i][companyCol]).trim() : "";
        break;
      }
    }

    if (rowIndex === -1) {
      return jsonResponse({ success: false, message: "Job not found" });
    }

    // Update ONLY the Status column
    sheet.getRange(rowIndex, statusCol + 1).setValue(finalStatus);

    // Identify company associated with the job and send notification
    var targetUserId = companyName;
    var usersSheet = ss.getSheetByName(USERS_SHEET_NAME || "Users");
    if (usersSheet && companyName) {
      var uData = usersSheet.getDataRange().getValues();
      if (uData.length > 1) {
        var uHeaders = uData[0].map(function (h) { return h.toString().trim().toLowerCase(); });
        var uIdCol = uHeaders.indexOf("user id");
        if (uIdCol === -1) uIdCol = uHeaders.indexOf("userid");
        if (uIdCol === -1) uIdCol = 0;
        var uNameCol = uHeaders.indexOf("name");
        if (uNameCol === -1) uNameCol = uHeaders.indexOf("company name");
        if (uNameCol === -1) uNameCol = 1;
        var uEmailCol = uHeaders.indexOf("email");
        if (uEmailCol === -1) uEmailCol = 2;

        for (var j = 1; j < uData.length; j++) {
          var uId = uData[j][uIdCol] ? String(uData[j][uIdCol]).trim() : "";
          var uName = uData[j][uNameCol] ? String(uData[j][uNameCol]).trim() : "";
          var uEmail = uData[j][uEmailCol] ? String(uData[j][uEmailCol]).trim() : "";
          if (uId.toLowerCase() === companyName.toLowerCase() ||
              uName.toLowerCase() === companyName.toLowerCase() ||
              uEmail.toLowerCase() === companyName.toLowerCase()) {
            targetUserId = uId;
            break;
          }
        }
      }
    }

    var notifTitle = "";
    var notifMsg = "";
    if (finalStatus === "Inactive") {
      notifTitle = "Job Deactivated by Admin";
      notifMsg = 'Your job "' + jobTitle + '" has been deactivated by the administrator.\nStudents cannot view or apply for this job.';
    } else {
      notifTitle = "Job Activated by Admin";
      notifMsg = 'Your job "' + jobTitle + '" has been activated by the administrator.\nStudents can now view and apply for this job.';
    }

    if (targetUserId) {
      addNotification(ss, targetUserId, notifTitle, notifMsg, "job_status");
      if (companyName && companyName.toLowerCase() !== targetUserId.toLowerCase()) {
        addNotification(ss, companyName, notifTitle, notifMsg, "job_status");
      }
    }

    return jsonResponse({
      success: true,
      message: "Job status updated to " + finalStatus + " successfully",
      jobId: foundJobId,
      status: finalStatus
    });
  } catch (err) {
    return jsonResponse({ success: false, message: "Failed to update job status: " + err.toString() });
  }
}

function handleGetMostDemandedSkills(ss, params) {
  try {
    var sheet = ss.getSheetByName("Jobs");
    if (!sheet) {
      return jsonResponse({ success: true, skills: [] });
    }
    var data = sheet.getDataRange().getValues();
    if (data.length <= 1) {
      return jsonResponse({ success: true, skills: [] });
    }

    var headers = data[0].map(function (h) { return h.toString().trim().toLowerCase(); });
    var skillsCol = headers.indexOf("skills");
    if (skillsCol === -1) skillsCol = headers.indexOf("requiredskills");
    if (skillsCol === -1) skillsCol = headers.indexOf("skillsrequired");
    if (skillsCol === -1) skillsCol = headers.indexOf("required skills");
    if (skillsCol === -1) skillsCol = 5;

    var skillCounts = {};
    var skillDisplayMap = {};

    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      if (!row[0] && !row[1]) continue;

      var status = "";
      var statusCol = headers.indexOf("status");
      if (statusCol === -1) statusCol = headers.indexOf("jobstatus");
      if (statusCol !== -1 && row[statusCol] !== undefined && row[statusCol] !== null) {
        status = row[statusCol].toString().trim().toLowerCase();
      }
      if (status && status !== "active" && status !== "open") {
        continue;
      }

      var rawSkills = (row[skillsCol] !== undefined && row[skillsCol] !== null)
        ? row[skillsCol].toString().trim()
        : "";

      if (!rawSkills) continue;

      var splitTokens = rawSkills.split(/[,/;\n|]/);
      for (var k = 0; k < splitTokens.length; k++) {
        var token = splitTokens[k].trim();
        if (token.length > 0) {
          var canonical = token.toLowerCase();
          if (!skillDisplayMap[canonical]) {
            skillDisplayMap[canonical] = token;
          }
          skillCounts[canonical] = (skillCounts[canonical] || 0) + 1;
        }
      }
    }

    var resultList = [];
    for (var key in skillCounts) {
      resultList.push({
        skill: skillDisplayMap[key] || key,
        count: skillCounts[key]
      });
    }

    resultList.sort(function (a, b) {
      return b.count - a.count;
    });

    return jsonResponse({ success: true, skills: resultList });
  } catch (err) {
    return jsonResponse({ success: false, message: "Error fetching demanded skills: " + err.toString() });
  }
}

      if (s.github) compScore += 7.5;
      if (s.linkedin) compScore += 7.5;

      var sSkillList = s.skills ? s.skills.split(/[,/;\n|]/).map(function(sk){ return sk.trim().toLowerCase(); }).filter(function(sk){ return sk.length > 0; }) : [];
      if (sSkillList.length >= 5) compScore += 25.0;
      else if (sSkillList.length >= 3) compScore += 18.0;
      else if (sSkillList.length >= 1) compScore += 10.0;

      if (s.projects) compScore += 15.0;
      if (s.certifications) compScore += 10.0;
      if (compScore > 100.0) compScore = 100.0;

      var mktScore = 100.0;
      if (activeJobs.length > 0) {
        var totalMatchSum = 0.0;
        for (var jIdx = 0; jIdx < activeJobs.length; jIdx++) {
          var jobSkillStr = activeJobs[jIdx];
          var reqSkills = jobSkillStr ? jobSkillStr.split(/[,/;\n|]/).map(function(sk){ return sk.trim().toLowerCase(); }).filter(function(sk){ return sk.length > 0; }) : [];
          if (reqSkills.length === 0) {
            totalMatchSum += 100.0;
          } else {
            var matched = 0;
            for (var r = 0; r < reqSkills.length; r++) {
              if (sSkillList.indexOf(reqSkills[r]) !== -1) {
                matched++;
              }
            }
            totalMatchSum += (matched / reqSkills.length) * 100.0;
          }
        }
        mktScore = totalMatchSum / activeJobs.length;
      }

      var overall = (compScore * 0.50) + (mktScore * 0.50);

      if (overall >= 80.0) {
        counts["Ready"]++;
      } else if (overall >= 50.0) {
        counts["Almost Ready"]++;
      } else {
        counts["Needs Improvement"]++;
      }
    }

    var categories = [
      { category: "Ready", count: counts["Ready"] },
      { category: "Almost Ready", count: counts["Almost Ready"] },
      { category: "Needs Improvement", count: counts["Needs Improvement"] }
    ];

    return jsonResponse({
      success: true,
      ready: counts["Ready"],
      almostReady: counts["Almost Ready"],
      needsImprovement: counts["Needs Improvement"],
      totalStudents: totalCount,
      categories: categories,
      distribution: {
        ready: counts["Ready"],
        almostReady: counts["Almost Ready"],
        needsImprovement: counts["Needs Improvement"],
        totalStudents: totalCount
      },
      readinessDistribution: {
        ready: counts["Ready"],
        almostReady: counts["Almost Ready"],
        needsImprovement: counts["Needs Improvement"],
        totalStudents: totalCount
      }
    });
  } catch (err) {
    return jsonResponse({ success: false, message: "Error calculating readiness distribution: " + err.toString() });
  }
}

function handleGetTopRecommendedJobs(ss, params) {
  try {
    var jobsSheet = ss.getSheetByName("Jobs");
    if (!jobsSheet) {
      return jsonResponse({ success: true, recommendedJobs: [], jobs: [] });
    }

    var jData = jobsSheet.getDataRange().getValues();
    if (jData.length <= 1) {
      return jsonResponse({ success: true, recommendedJobs: [], jobs: [] });
    }

    var jHeaders = jData[0].map(function (h) { return h.toString().trim().toLowerCase(); });
    var getJVal = function (row, names) {
      for (var k = 0; k < names.length; k++) {
        var idx = jHeaders.indexOf(names[k].toLowerCase());
        if (idx !== -1 && row[idx] !== undefined && row[idx] !== null) {
          return row[idx].toString().trim();
        }
      }
      return "";
    };

    var jobsList = [];
    for (var j = 1; j < jData.length; j++) {
      var jRow = jData[j];
      var jId = getJVal(jRow, ["jobid", "id", "job id"]);
      var status = getJVal(jRow, ["status", "jobstatus"]);
      if (status && status.toLowerCase() !== "active" && status.toLowerCase() !== "open") {
        continue;
      }
      if (jId || getJVal(jRow, ["title", "jobtitle", "job title"])) {
        jobsList.push({
          jobId: jId || ("JOB" + j),
          title: getJVal(jRow, ["title", "jobtitle", "job title"]),
          company: getJVal(jRow, ["company", "companyname", "company name"]),
          skills: getJVal(jRow, ["skills", "requiredskills", "skillsrequired"])
        });
      }
    }

    var studentSkillsList = [];
    var profSheet = ss.getSheetByName("StudentProfiles");
    if (profSheet) {
      var pData = profSheet.getDataRange().getValues();
      if (pData.length > 1) {
        var pHeaders = pData[0].map(function (h) { return h.toString().trim().toLowerCase(); });
        var sCol = pHeaders.indexOf("skills");
        for (var p = 1; p < pData.length; p++) {
          var pSkills = (sCol !== -1 && pData[p][sCol]) ? pData[p][sCol].toString().trim() : "";
          var tokens = pSkills.split(/[,/;\n|]/).map(function(sk){ return sk.trim().toLowerCase(); }).filter(function(sk){ return sk.length > 0; });
          studentSkillsList.push(tokens);
        }
      }
    }

    var recCounts = {};
    var avgMatchMap = {};
    var missingSkillsMap = {};

    for (var i = 0; i < jobsList.length; i++) {
      var job = jobsList[i];
      var reqSkills = job.skills.split(/[,/;\n|]/).map(function(sk){ return sk.trim().toLowerCase(); }).filter(function(sk){ return sk.length > 0; });

      var count = 0;
      var totalMatchPct = 0;

      for (var s = 0; s < studentSkillsList.length; s++) {
        var stSkills = studentSkillsList[s];
        if (reqSkills.length === 0) {
          count++;
          totalMatchPct += 100;
        } else {
          var matched = 0;
          for (var r = 0; r < reqSkills.length; r++) {
            if (stSkills.indexOf(reqSkills[r]) !== -1) {
              matched++;
            }
          }
          var matchPct = (matched / reqSkills.length) * 100.0;
          totalMatchPct += matchPct;
          if (matchPct >= 50.0) {
            count++;
          }
        }
      }

      recCounts[job.jobId] = count;
      avgMatchMap[job.jobId] = studentSkillsList.length > 0 ? Math.round(totalMatchPct / studentSkillsList.length) : 0;
      missingSkillsMap[job.jobId] = reqSkills.join(", ");
    }

    var resultList = [];
    for (var k = 0; k < jobsList.length; k++) {
      var jb = jobsList[k];
      resultList.push({
        jobId: jb.jobId,
        jobTitle: jb.title,
        title: jb.title,
        company: jb.company,
        recommendationCount: recCounts[jb.jobId] || 0,
        matchPercentage: avgMatchMap[jb.jobId] || 0,
        missingSkills: missingSkillsMap[jb.jobId] || ""
      });
    }

    resultList.sort(function (a, b) {
      return b.recommendationCount - a.recommendationCount;
    });

    return jsonResponse({
      success: true,
      recommendedJobs: resultList,
      jobs: resultList
    });
  } catch (err) {
    return jsonResponse({ success: false, message: "Error fetching recommended jobs: " + err.toString() });
  }
}

function handleGetPlacementTrends(ss, params) {
  try {
    var appSheet = ss.getSheetByName("Applications");
    if (!appSheet) {
      return jsonResponse({ success: true, trends: [] });
    }

    var aData = appSheet.getDataRange().getValues();
    if (aData.length <= 1) {
      return jsonResponse({ success: true, trends: [] });
    }

    var aHeaders = aData[0].map(function (h) { return h.toString().trim().toLowerCase(); });
    var dateCol = aHeaders.indexOf("applied date");
    if (dateCol === -1) dateCol = aHeaders.indexOf("applieddate");
    if (dateCol === -1) dateCol = aHeaders.indexOf("date");
    if (dateCol === -1) dateCol = 3;

    var statusCol = aHeaders.indexOf("status");
    if (statusCol === -1) statusCol = aHeaders.indexOf("applicationstatus");
    if (statusCol === -1) statusCol = 4;

    var monthMap = {};

    var monthNames = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];

    for (var i = 1; i < aData.length; i++) {
      var row = aData[i];
      if (!row[0] && !row[1] && !row[2]) continue;

      var rawDate = row[dateCol];
      var dateObj = null;

      if (rawDate instanceof Date && !isNaN(rawDate.getTime())) {
        dateObj = rawDate;
      } else if (rawDate) {
        var strDate = rawDate.toString().trim();
        var parsed = Date.parse(strDate);
        if (!isNaN(parsed)) {
          dateObj = new Date(parsed);
        } else {
          var parts = strDate.split(" ")[0].split("-");
          if (parts.length >= 3) {
            var y = parseInt(parts[0], 10);
            var m = parseInt(parts[1], 10) - 1;
            var d = parseInt(parts[2], 10);
            if (!isNaN(y) && !isNaN(m) && !isNaN(d)) {
              dateObj = new Date(y, m, d);
            }
          }
        }
      }

      if (!dateObj) continue;

      var year = dateObj.getFullYear();
      var monthIdx = dateObj.getMonth();
      var sortKey = year + "-" + (monthIdx < 9 ? "0" + (monthIdx + 1) : (monthIdx + 1));
      var displayMonth = monthNames[monthIdx] + " " + year;

      if (!monthMap[sortKey]) {
        monthMap[sortKey] = {
          sortKey: sortKey,
          month: displayMonth,
          applications: 0,
          shortlisted: 0,
          placements: 0,
          selected: 0
        };
      }

      monthMap[sortKey].applications++;

      var status = (statusCol !== -1 && row[statusCol]) ? row[statusCol].toString().trim().toLowerCase() : "";
      if (status.indexOf("shortlist") !== -1) {
        monthMap[sortKey].shortlisted++;
      }
      if (status.indexOf("select") !== -1 || status.indexOf("hired") !== -1 || status.indexOf("accept") !== -1 || status.indexOf("place") !== -1) {
        monthMap[sortKey].placements++;
        monthMap[sortKey].selected++;
      }
    }

    var keys = [];
    for (var k in monthMap) {
      keys.push(k);
    }
    keys.sort();

    var trends = [];
    for (var m = 0; m < keys.length; m++) {
      trends.push({
        month: monthMap[keys[m]].month,
        applications: monthMap[keys[m]].applications,
        shortlisted: monthMap[keys[m]].shortlisted,
        placements: monthMap[keys[m]].placements,
        selected: monthMap[keys[m]].selected
      });
    }

    return jsonResponse({ success: true, trends: trends });
  } catch (err) {
    return jsonResponse({ success: false, message: "Error calculating placement trends: " + err.toString() });
  }
}

function handleSendBroadcast(spreadsheet, params) {
  try {
    var title = String(params.title || "").trim();
    var message = String(params.message || "").trim();
    var audience = String(params.audience || params.targetAudience || "all").trim().toLowerCase();
    var createdBy = String(params.createdBy || params.author || "Admin").trim();

    if (!title || !message) {
      return jsonResponse({
        success: false,
        message: "Title and message are required for broadcast."
      });
    }

    var sheet = spreadsheet.getSheetByName("Announcements");
    if (!sheet) {
      sheet = spreadsheet.insertSheet("Announcements");
      sheet.appendRow([
        "Broadcast ID",
        "Title",
        "Message",
        "Audience",
        "Created By",
        "Date",
        "Status"
      ]);
    }

    var broadcastId = "BC" + new Date().getTime();
    var tz = Session.getScriptTimeZone() || "GMT";
    var dateStr = Utilities.formatDate(new Date(), tz, "yyyy-MM-dd HH:mm");

    sheet.appendRow([
      broadcastId,
      title,
      message,
      audience,
      createdBy,
      dateStr,
      "Active"
    ]);

    // Push individual notifications to targeted users in Notifications sheet
    try {
      var usersSheet = spreadsheet.getSheetByName("Users");
      if (usersSheet) {
        var uData = usersSheet.getDataRange().getValues();
        if (uData.length > 1) {
          var uHeaders = uData[0].map(function (h) { return h.toString().trim().toLowerCase(); });
          var uIdCol = uHeaders.indexOf("user id");
          if (uIdCol === -1) uIdCol = uHeaders.indexOf("userid");
          if (uIdCol === -1) uIdCol = 0;

          var roleCol = uHeaders.indexOf("role");
          if (roleCol === -1) roleCol = uHeaders.indexOf("user role");
          if (roleCol === -1) roleCol = 4;

          for (var i = 1; i < uData.length; i++) {
            var row = uData[i];
            var uId = row[uIdCol] ? String(row[uIdCol]).trim() : "";
            var uRole = (roleCol !== -1 && row[roleCol]) ? String(row[roleCol]).trim().toLowerCase() : "";
            if (!uId) continue;

            var matches = false;
            if (audience === "all" || audience === "" || audience === "all users") {
              matches = true;
            } else if ((audience === "student" || audience === "students") && (uRole === "student" || uRole === "student user" || uId.toUpperCase().indexOf("STU") !== -1)) {
              matches = true;
            } else if ((audience === "company" || audience === "companies" || audience === "recruiter" || audience === "recruiters") && (uRole === "company" || uRole === "recruiter" || uId.toUpperCase().indexOf("COM") !== -1)) {
              matches = true;
            }

            if (matches) {
              addNotification(spreadsheet, uId, "Broadcast: " + title, message, "broadcast");
            }
          }
        }
      }
    } catch (notifErr) {
      Logger.log("Failed to dispatch notifications for broadcast: " + notifErr.toString());
    }

    return jsonResponse({
      success: true,
      message: "Broadcast announcement sent successfully.",
      broadcastId: broadcastId
    });

  } catch (error) {
    return jsonResponse({
      success: false,
      message: "Failed to send broadcast: " + error.toString()
    });
  }
}

        var mobileCol = uHeaders.indexOf("mobile");
        if (mobileCol === -1) mobileCol = uHeaders.indexOf("phone");
        if (mobileCol === -1) mobileCol = 3;

        var statusCol = uHeaders.indexOf("status");
        if (statusCol === -1) statusCol = uHeaders.indexOf("user status");
        if (statusCol === -1) statusCol = 5;

        for (var i = 1; i < uData.length; i++) {
          if (uData[i][uIdCol] && String(uData[i][uIdCol]).trim().toLowerCase() === userId.toLowerCase()) {
            if (name !== null) usersSheet.getRange(i + 1, nameCol + 1).setValue(name);
            if (email !== null) usersSheet.getRange(i + 1, emailCol + 1).setValue(email);
            if (mobile !== null) usersSheet.getRange(i + 1, mobileCol + 1).setValue(mobile);
            if (status !== null) usersSheet.getRange(i + 1, statusCol + 1).setValue(status);
            updatedInUsers = true;
            break;
          }
        }
      }
    }

    var updatedInProfiles = false;
    var profSheet = spreadsheet.getSheetByName("StudentProfiles");
    if (profSheet) {
      var pData = profSheet.getDataRange().getValues();
      if (pData.length > 1) {
        var pHeaders = pData[0].map(function (h) { return h.toString().trim().toLowerCase(); });
        var pIdCol = pHeaders.indexOf("user id");
        if (pIdCol === -1) pIdCol = pHeaders.indexOf("userid");
        if (pIdCol === -1) pIdCol = pHeaders.indexOf("student id");
        if (pIdCol === -1) pIdCol = 0;

        var pNameCol = pHeaders.indexOf("name");
        if (pNameCol === -1) pNameCol = pHeaders.indexOf("fullname");
        if (pNameCol === -1) pNameCol = 1;

        var pEmailCol = pHeaders.indexOf("email");
        if (pEmailCol === -1) pEmailCol = pHeaders.indexOf("emailaddress");
        if (pEmailCol === -1) pEmailCol = 2;

        var pMobileCol = pHeaders.indexOf("mobile");
        if (pMobileCol === -1) pMobileCol = pHeaders.indexOf("phone");
        if (pMobileCol === -1) pMobileCol = 3;

        var pEduCol = pHeaders.indexOf("education");
        if (pEduCol === -1) pEduCol = pHeaders.indexOf("course");
        if (pEduCol === -1) pEduCol = 4;

        var pSemCol = pHeaders.indexOf("semester");
        if (pSemCol === -1) pSemCol = pHeaders.indexOf("sem");
        if (pSemCol === -1) pSemCol = 5;

        var pSkillsCol = pHeaders.indexOf("skills");
        if (pSkillsCol === -1) pSkillsCol = pHeaders.indexOf("skill");
        if (pSkillsCol === -1) pSkillsCol = 6;

        for (var k = 1; k < pData.length; k++) {
          if (pData[k][pIdCol] && String(pData[k][pIdCol]).trim().toLowerCase() === userId.toLowerCase()) {
            if (name !== null) profSheet.getRange(k + 1, pNameCol + 1).setValue(name);
            if (email !== null) profSheet.getRange(k + 1, pEmailCol + 1).setValue(email);
            if (mobile !== null) profSheet.getRange(k + 1, pMobileCol + 1).setValue(mobile);
            if (education !== null && pEduCol !== -1) profSheet.getRange(k + 1, pEduCol + 1).setValue(education);
            if (semester !== null && pSemCol !== -1) profSheet.getRange(k + 1, pSemCol + 1).setValue(semester);
            if (skills !== null && pSkillsCol !== -1) profSheet.getRange(k + 1, pSkillsCol + 1).setValue(skills);
            updatedInProfiles = true;
            break;
          }
        }
      }
    }

    if (!updatedInUsers && !updatedInProfiles) {
      return jsonResponse({
        success: false,
        message: "Student record (" + userId + ") was not found."
      });
    }

    return jsonResponse({
      success: true,
      message: "Student record (" + userId + ") updated successfully."
    });

  } catch (error) {
    return jsonResponse({
      success: false,
      message: "Failed to update student: " + error.toString()
    });
  }
}
